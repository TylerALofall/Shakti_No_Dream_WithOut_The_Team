/*
 * Spiral.Template_make.c  ->  shakti-template-make
 *
 * Whole-file template extractor, Tyler's spiral law.
 * Static BSS memory edition (Zero malloc/realloc/free).
 *
 * Usage: shakti-template-make <file> [full]
 * Exit:  0 clean, 1 usage/io/memory, 2 unbalanced record (bench).
 *
 * C99 strict: cc -std=c99 -pedantic -Wall -Wextra -Werror -O2
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

/* ---------- BSS Capacity Definitions ---------- */

#define LINE_BUF_CAP  (20 * 1024 * 1024)  /* 20 MB static line buffer */
#define MAX_TAG_LEN   256                 /* Hard ceiling on tag names */
#define MAX_NODES     100000              /* Static node arena capacity */
#define MAX_CHILDREN  1024                /* Max children per parent node */
#define MAX_ROOTS     10000               /* Max distinct roots/forms */
#define MAX_STACK     4096                /* Max tree depth stack */

/* ---------- BSS Static Line Reader ---------- */

static char g_line_buf[LINE_BUF_CAP];

static char *read_line_static(FILE *fp)
{
    size_t n = 0;
    int c;
    while ((c = fgetc(fp)) != EOF) {
        if (n + 1 >= LINE_BUF_CAP - 1) {
            fprintf(stderr, "BENCH: Line buffer exceeded 20MB limit\n");
            return NULL;
        }
        g_line_buf[n++] = (char)c;
        if (c == '\n') break;
    }
    if (n == 0 && c == EOF) return NULL;
    g_line_buf[n] = '\0';
    return g_line_buf;
}

/* ---------- Static Node Arena ---------- */

typedef struct Node {
    char tag[MAX_TAG_LEN];
    long count;
    int child[MAX_CHILDREN];
    int nchild;
} Node;

static Node g_node_arena[MAX_NODES];
static int g_arena_top = 0;

static int node_new(const char *tag)
{
    if (g_arena_top >= MAX_NODES) {
        fprintf(stderr, "BENCH: Node arena exhausted (%d nodes)\n", MAX_NODES);
        return -1;
    }
    int idx = g_arena_top++;
    strncpy(g_node_arena[idx].tag, tag, MAX_TAG_LEN - 1);
    g_node_arena[idx].tag[MAX_TAG_LEN - 1] = '\0';
    g_node_arena[idx].count = 1;
    g_node_arena[idx].nchild = 0;
    return idx;
}

static int same_signature(int a_idx, int b_idx)
{
    int i;
    const Node *a = &g_node_arena[a_idx];
    const Node *b = &g_node_arena[b_idx];
    if (strcmp(a->tag, b->tag) != 0) return 0;
    if (a->nchild != b->nchild) return 0;
    for (i = 0; i < a->nchild; i++) {
        if (!same_signature(a->child[i], b->child[i])) return 0;
    }
    return 1;
}

/* attach child under parent; collapse into an identical sibling shape */
static int attach(int parent_idx, int child_idx)
{
    int i;
    Node *parent = &g_node_arena[parent_idx];
    for (i = 0; i < parent->nchild; i++) {
        if (same_signature(parent->child[i], child_idx)) {
            g_node_arena[parent->child[i]].count += g_node_arena[child_idx].count;
            return 0;
        }
    }
    if (parent->nchild >= MAX_CHILDREN) {
        fprintf(stderr, "BENCH: Max children (%d) exceeded\n", MAX_CHILDREN);
        return -1;
    }
    parent->child[parent->nchild++] = child_idx;
    return 0;
}

/* ---------- Static Forest ---------- */

static int g_roots[MAX_ROOTS];
static int g_nroots = 0;

static int forest_add(int root_idx)
{
    int i;
    for (i = 0; i < g_nroots; i++) {
        if (same_signature(g_roots[i], root_idx)) {
            g_node_arena[g_roots[i]].count += g_node_arena[root_idx].count;
            return 0;
        }
    }
    if (g_nroots >= MAX_ROOTS) {
        fprintf(stderr, "BENCH: Max roots (%d) exceeded\n", MAX_ROOTS);
        return -1;
    }
    g_roots[g_nroots++] = root_idx;
    return 0;
}

/* ---------- Tag Walking ---------- */

static int next_tag(const char *line, size_t *pos,
                    char *tag, size_t cap, int *is_close, int *kind)
{
    size_t i = *pos, start, n;
    while (line[i] && line[i] != '<') i++;
    if (line[i] != '<') {
        *pos = i;
        return 0;
    }
    i++;
    *is_close = 0;
    *kind = 0;
    if (line[i] == '?') {
        *kind = 1;
        while (line[i] && !(line[i] == '?' && line[i + 1] == '>')) i++;
        if (line[i]) i += 2;
        *pos = i;
        return 1;
    }
    if (line[i] == '!') {
        *kind = (line[i + 1] == '-' && line[i + 2] == '-') ? 2 : 3;
        while (line[i] && line[i] != '>') i++;
        if (line[i] == '>') i++;
        *pos = i;
        return 1;
    }
    if (line[i] == '/') { *is_close = 1; i++; }
    start = i;
    while (line[i] && line[i] != '>' && !isspace((unsigned char)line[i])
           && line[i] != '/') i++;
    n = i - start;

    /* If tag length is invalid or exceeds buffer cap, advance *pos past '<' */
    if (n == 0 || n >= cap) {
        *pos = start;
        return 0;
    }

    memcpy(tag, line + start, n);
    tag[n] = '\0';
    while (line[i] && line[i] != '>') i++;
    if (line[i] == '>') i++;
    *pos = i;
    return 1;
}

static int closes_on_line(const char *line, size_t *pos, const char *tag)
{
    size_t p = *pos;
    char t[MAX_TAG_LEN];
    int is_close = 0, kind = 0;
    long interim = 0;
    while (next_tag(line, &p, t, sizeof t, &is_close, &kind)) {
        if (kind != 0) continue;
        if (!is_close && strcmp(t, tag) == 0) {
            size_t q = p;
            if (q >= 2 && line[q - 2] == '/') continue;
            interim++;
        } else if (is_close && strcmp(t, tag) == 0) {
            if (interim == 0) { *pos = p; return 1; }
            interim--;
        }
    }
    return 0;
}

static int line_blank(const char *s)
{
    while (*s) {
        if (!isspace((unsigned char)*s)) return 0;
        s++;
    }
    return 1;
}

static int line_is(const char *s, const char *word)
{
    while (isspace((unsigned char)*s)) s++;
    while (*word) {
        if (*s != *word) return 0;
        s++; word++;
    }
    return *s == '\0' || *s == '\n' || isspace((unsigned char)*s);
}

/* ---------- Emission Laws ---------- */

static int g_section_idx = -1;
static long g_repeats = 0;
static int g_variants = 0;
static int g_sec_lvl = 0;
static int g_full = 0;

static int find_section(int node_idx, int is_root)
{
    int i;
    const Node *n = &g_node_arena[node_idx];
    if (!is_root && n->count > 1) return node_idx;
    for (i = 0; i < n->nchild; i++) {
        int r = find_section(n->child[i], 0);
        if (r != -1) return r;
    }
    return -1;
}

static int count_same_tag(int parent_idx, const char *tag)
{
    int i, k = 0;
    const Node *parent = &g_node_arena[parent_idx];
    for (i = 0; i < parent->nchild; i++)
        if (strcmp(g_node_arena[parent->child[i]].tag, tag) == 0) k++;
    return k;
}

static int find_parent(int node_idx, int target_idx, int lvl, int *found_lvl)
{
    int i;
    const Node *n = &g_node_arena[node_idx];
    for (i = 0; i < n->nchild; i++) {
        if (n->child[i] == target_idx) { *found_lvl = lvl + 1; return node_idx; }
        {
            int r = find_parent(n->child[i], target_idx, lvl + 1, found_lvl);
            if (r != -1) return r;
        }
    }
    return -1;
}

static void lower_copy(char *dst, size_t cap, const char *src)
{
    size_t i;
    for (i = 0; src[i] && i + 1 < cap; i++)
        dst[i] = (char)tolower((unsigned char)src[i]);
    dst[i] = '\0';
}

static void join3(char *dst, size_t cap, const char *a, const char *b)
{
    size_t na = strlen(a), nb = strlen(b);
    if (na + 1 + nb >= cap) { dst[0] = '\0'; return; }
    memcpy(dst, a, na);
    dst[na] = '.';
    memcpy(dst + na + 1, b, nb + 1);
}

static void emit_node(int node_idx, int lvl, const char *prefix)
{
    int i;
    const Node *n = &g_node_arena[node_idx];
    if (n->nchild == 0) {
        char p[512];
        join3(p, sizeof p, prefix, n->tag);
        printf("<%s lvl=\"%d\">[%s]</%s>\n", n->tag, lvl, p, n->tag);
        return;
    }
    printf("<%s lvl=\"%d\">\n", n->tag, lvl);
    for (i = 0; i < n->nchild; i++) {
        int c_idx = n->child[i];
        const Node *c = &g_node_arena[c_idx];
        const Node *sec = g_section_idx != -1 ? &g_node_arena[g_section_idx] : NULL;

        int wraps = (c_idx == g_section_idx) ||
                    (g_full && sec && strcmp(c->tag, sec->tag) == 0);
        if (sec && !g_full && c_idx != g_section_idx && strcmp(c->tag, sec->tag) == 0)
            continue;
        if (wraps) {
            if (g_sec_lvl == 0) g_sec_lvl = lvl + 1;
            printf("<SECTION lvl=\"%d\" id=\"%s\">\n", lvl + 1, c->tag);
            if (c->nchild == 0) {
                emit_node(c_idx, lvl + 2, "section");
            } else {
                int j;
                for (j = 0; j < c->nchild; j++) {
                    int g_idx = c->child[j];
                    const Node *g = &g_node_arena[g_idx];
                    if (g->nchild == 0)
                        emit_node(g_idx, lvl + 2, "section");
                    else
                        emit_node(g_idx, lvl + 2, g->tag);
                }
            }
            printf("</SECTION>\n");
            continue;
        }
        if (c->nchild == 0)
            emit_node(c_idx, lvl + 1, prefix);
        else
            emit_node(c_idx, lvl + 1, c->tag);
    }
    printf("</%s>\n", n->tag);
}

static void emit_template(int root_idx)
{
    char rootlow[256];
    const Node *root = &g_node_arena[root_idx];
    lower_copy(rootlow, sizeof rootlow, root->tag);

    g_section_idx = find_section(root_idx, 1);
    g_repeats = 0;
    g_variants = 0;
    g_sec_lvl = 0;
    if (g_section_idx != -1) {
        int slvl = 0;
        int par_idx = find_parent(root_idx, g_section_idx, 1, &slvl);
        g_repeats = g_node_arena[g_section_idx].count;
        if (par_idx != -1) 
            g_variants = count_same_tag(par_idx, g_node_arena[g_section_idx].tag) - 1;
    }

    if (root->nchild == 0) {
        char p[512];
        join3(p, sizeof p, rootlow, root->tag);
        printf("<%s lvl=\"0\">[%s]</%s>\n", root->tag, p, root->tag);
        return;
    }
    emit_node(root_idx, 0, rootlow);

    if (g_section_idx != -1) {
        printf("<!-- two-match CONFIRMED: <SECTION> level=%d repeats=%ld;"
               " repeat written once;", g_sec_lvl, g_repeats);
        if (g_variants > 0 && !g_full)
            printf(" %d later variant(s) unseen (run 'full' to override)",
                   g_variants);
        else
            printf(" no variants");
        printf(" -->\n");
    }
}

/* ---------- Main Execution ---------- */

int main(int argc, char **argv)
{
    FILE *fp;
    char *line;
    long lineno = 0;
    int in_comment = 0;
    int rc = 0;
    int fence_mode = -1;
    int in_fence = 0;
    long documents = 0;

    static int stack[MAX_STACK];
    int sp = 0;
    int cur_root = -1;

    if (argc < 2 || argc > 3) {
        fprintf(stderr, "usage: shakti-template-make <file> [full]\n");
        return 1;
    }
    if (argc == 3) {
        if (strcmp(argv[2], "full") != 0) {
            fprintf(stderr, "usage: shakti-template-make <file> [full]\n");
            return 1;
        }
        g_full = 1;
    }

    {
        size_t L = strlen(argv[1]);
        if (L >= 3 && strcmp(argv[1] + L - 3, ".md") == 0)
            fence_mode = 1;
    }

    fp = fopen(argv[1], "r");
    if (!fp) {
        fprintf(stderr, "cannot open: %s\n", argv[1]);
        return 1;
    }

    while ((line = read_line_static(fp)) != NULL) {
        size_t pos = 0;
        char tag[MAX_TAG_LEN];
        int is_close = 0, kind = 0;
        lineno++;

        if (fence_mode != 0 && line_is(line, "```xml")) {
            if (sp == 0 && cur_root == -1) { fence_mode = 1; in_fence = 1; }
            continue;
        }
        if (fence_mode == 1 && in_fence && line_is(line, "```")) {
            in_fence = 0;
            continue;
        }
        if (fence_mode == 1 && !in_fence) continue;
        if (fence_mode == -1) {
            size_t p0 = 0;
            char t0[MAX_TAG_LEN];
            int c0 = 0, k0 = 0;
            if (!next_tag(line, &p0, t0, sizeof t0, &c0, &k0)) continue;
            fence_mode = 0;
        }

        if (line_blank(line)) continue;

        if (in_comment) {
            char *e = strstr(line, "-->");
            if (!e) continue;
            in_comment = 0;
            pos = (size_t)(e - line) + 3;
        }

        while (next_tag(line, &pos, tag, sizeof tag, &is_close, &kind)) {
            if (kind == 1 || kind == 3) continue;
            if (kind == 2) {
                char *e = strstr(line + pos, "-->");
                if (e) pos = (size_t)(e - line) + 3;
                else { in_comment = 1; break; }
                continue;
            }
            if (!is_close && pos >= 2 && line[pos - 2] == '/') {
                int n = node_new(tag);
                if (n < 0 || sp < 1 || attach(stack[sp - 1], n) != 0) {
                    fprintf(stderr, "BENCH: attach failed line %ld\n", lineno);
                    rc = 1; break;
                }
            } else if (!is_close && closes_on_line(line, &pos, tag)) {
                int n = node_new(tag);
                if (n < 0 || sp < 1 || attach(stack[sp - 1], n) != 0) {
                    fprintf(stderr, "BENCH: attach failed line %ld\n", lineno);
                    rc = 1; break;
                }
            } else if (!is_close) {
                int n = node_new(tag);
                if (n < 0) { fprintf(stderr, "BENCH: memory\n"); rc = 1; break; }
                if (sp >= MAX_STACK) {
                    fprintf(stderr, "BENCH: stack overflow at line %ld\n", lineno);
                    rc = 1; break;
                }
                if (sp == 0) {
                    if (cur_root != -1) {
                        fprintf(stderr, "BENCH: unclosed root before line %ld\n", lineno);
                        rc = 2; break;
                    }
                    cur_root = n;
                }
                stack[sp++] = n;
            } else {
                int done;
                if (sp <= 0 || strcmp(g_node_arena[stack[sp - 1]].tag, tag) != 0) {
                    fprintf(stderr,
                            "BENCH: line %ld closes %s but expected %s\n",
                            lineno, tag,
                            sp > 0 ? g_node_arena[stack[sp - 1]].tag
                                   : "(nothing open)");
                    rc = 2; break;
                }
                done = stack[--sp];
                if (sp == 0) {
                    cur_root = -1;
                    documents++;
                    if (forest_add(done) != 0) {
                        fprintf(stderr, "BENCH: memory\n");
                        rc = 1; break;
                    }
                } else if (attach(stack[sp - 1], done) != 0) {
                    fprintf(stderr, "BENCH: attach failed line %ld\n", lineno);
                    rc = 1; break;
                }
            }
        }
        if (rc != 0) break;
    }
    fclose(fp);

    if (rc == 0 && sp != 0) {
        fprintf(stderr, "BENCH: unbalanced, %d record(s) never closed\n", sp);
        rc = 2;
    }
    if (rc == 0 && g_nroots == 0) {
        fprintf(stderr, "BENCH: no records found\n");
        rc = 2;
    }

    if (rc == 0) {
        int i;
        for (i = 0; i < g_nroots; i++) {
            int same = 0, j, form = 0;
            for (j = 0; j < g_nroots; j++)
                if (strcmp(g_node_arena[g_roots[j]].tag, g_node_arena[g_roots[i]].tag) == 0) same++;
            for (j = 0; j < i; j++)
                if (strcmp(g_node_arena[g_roots[j]].tag, g_node_arena[g_roots[i]].tag) == 0) form++;
            if (same > 1)
                printf("<!-- <%s> form %d of %d: repeats=%ld -->\n",
                       g_node_arena[g_roots[i]].tag, form + 1, same,
                       g_node_arena[g_roots[i]].count);
            emit_template(g_roots[i]);
            putchar('\n');
        }
        printf("<!-- documents walked: %ld; distinct forms: %d -->\n",
               documents, g_nroots);
    }

    return rc;
}
